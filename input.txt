int m() {
    int i, j, k;
    int limit, result;
    float x, y, val;

    limit = 10;
    result = 0;
    x = 1.0;
    y = 2.5;

    /* Testing Complex Arithmetic and Unary operators */
    val = (x + y) * 5 / 2 - (-x);
    
    /* Testing Math Functions and Argument Lists */
    val = pow(x, y) + sqrt(16.0);

    /* Testing Nested Loops (FOR/WHILE) */
    for (i = 0; i < limit; i++) {
        j = 0;
        while (j < 5) {
            /* Testing Conditionals with Logic operators */
            if (i > 5 && j != 0) {
                result = result + i * j;
            } else {
                result = result + 1;
            }
            
            /* Testing Increment/Decrement */
            j++;
        }
    }

    /* Testing Logical NOT and RELOP */
    if (!(result < 100)) {
        println(result);
    }

    return result;
}
